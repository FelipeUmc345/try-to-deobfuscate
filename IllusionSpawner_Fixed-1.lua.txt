local function safeRequire(obj)
	if not obj then return nil end
	local ok, result = pcall(require, obj)

-- // HELPERS // --

local function formatNumber(num)
	local suffixes = {"","K","M","B","T","Q"}
	local tier = 1
	while num >= 1000 and tier < #suffixes do num = num/1000; tier = tier+1 end
	if num % 1 == 0 then return string.format("%d%s", num, suffixes[tier])
	else return string.format("%.1f%s", num, suffixes[tier]) end
end

local function parseCurrency(str)
	str = str:gsub("<[^>]+>",""):gsub("%$",""):gsub(",",""):gsub("[^%d%.%a]","")
	local numStr, suffix = str:match("^(%d*%.?%d+)(%a*)")
	local num = tonumber(numStr) or 0
	suffix = suffix:upper()
	local mult = {K=1e3,M=1e6,B=1e9,T=1e12,Q=1e15}
	return num * (mult[suffix] or 1)
end

-- Safe cashout: doesn't WaitForChild (avoids hangs)
local function showCashout(text, isGreen, playSound)
	pcall(function()
		local lb = PlayerGui:FindFirstChild("LeftBottom")
		if not lb then return end
		local inner = lb:FindFirstChild("LeftBottom")
		if not inner then return end
		local CashoutFrame = inner:FindFirstChild("Cashout")
		if not CashoutFrame then return end
		local template = CashoutFrame:FindFirstChild("Template")
		if not template or not template:IsA("TextLabel") then return end
		local label = template:Clone()
		label.Text = text
		label.TextColor3 = isGreen and Color3.fromRGB(13,255,0) or Color3.fromRGB(255,0,0)
		label.Visible = true
		label.Parent = CashoutFrame
		local tweenInfo = TweenInfo.new(5, Enum.EasingStyle.Linear)
		local t1 = TweenService:Create(label, tweenInfo, {TextTransparency=1})
		local stroke = label:FindFirstChildOfClass("UIStroke")
		if stroke then TweenService:Create(stroke, tweenInfo, {Transparency=1}):Play() end
		t1.Completed:Once(function() label:Destroy() end)
		t1:Play()
		if playSound and isGreen and SoundController then
			pcall(function() SoundController:PlaySound("Sounds.Sfx.Cashout") end)
		end
	end)
end

local function getTraitIconByName(traitName)
	for assetId, data in pairs(traitMap) do
		if data.name == traitName then return assetId end
	end
	return nil
end

local function applyMutationVisuals(model, mutationName)
	if not mutationName or mutationName == "None" then return end
	local mutData = mutationMap[mutationName]
	if not mutData then return end
	pcall(function()
		local MutationsData = ReplicatedStorage.Datas:FindFirstChild("Mutations")
		if not MutationsData then return end
		local mutModule = safeRequire(MutationsData)
		if not mutModule then return end
		local mutInfo = mutModule[mutationName]
		if mutInfo and mutInfo.Palettes then
			local palette = mutInfo.Palettes[1]
			if palette then
				local colorIndex = 1
				for _, part in ipairs(model:GetDescendants()) do
					if part:IsA("BasePart") or part:IsA("MeshPart") then
						part.Color = palette[colorIndex]
						colorIndex = (colorIndex % #palette) + 1
					end
				end
			end
		end
	end)
end

-- // SPAWN ANIMAL // --
local spawnedIllusions = {}

local function spawnAnimal(animalName, traits, mutation, targetPlayerName)
	if not Animals or not Animals[animalName] then
		warn("[Spawn] Animal não encontrado nos dados: " .. tostring(animalName))
		return false, "invalid animal"
	end
	local animalData = Animals[animalName]

	-- Try to find model
	local modelsFolder = ReplicatedStorage:FindFirstChild("Models")
	local animalsFolder = modelsFolder and modelsFolder:FindFirstChild("Animals")
	local animalModel = animalsFolder and animalsFolder:FindFirstChild(animalName)
	if not animalModel then
		warn("[Spawn] Modelo não encontrado: " .. tostring(animalName))
		return false, "model not found"
	end

	-- Find plot
	local function getPlotForPlayer(playerName)
		local plotsFolder = workspace:FindFirstChild("Plots")
		if not plotsFolder then return nil end
		for _, plot in ipairs(plotsFolder:GetChildren()) do
			-- method 1: PlotSign surface GUI
			local plotSign = plot:FindFirstChild("PlotSign")
			if plotSign then
				local surfaceGui = plotSign:FindFirstChild("SurfaceGui")
				if surfaceGui then
					local frame = surfaceGui:FindFirstChild("Frame")
					if frame then
						for _, child in ipairs(frame:GetDescendants()) do
							if child:IsA("TextLabel") then
								local t = child.Text:gsub("^%s*(.-)%s*$","%1")
								if t:lower() == playerName:lower() then return plot end
							end
						end
					end
				end
			end
			-- method 2: plot named after player or has owner value
			local owner = plot:FindFirstChild("Owner")
			if owner and tostring(owner.Value):lower() == playerName:lower() then
				return plot
			end
		end
		return nil
	end

	local playerName = targetPlayerName or LocalPlayer.Name
	local targetPlot = getPlotForPlayer(playerName)
	if not targetPlot then
		warn("[Spawn] Plot não encontrado para: " .. tostring(playerName))
		return false, "plot not found"
	end

	local podiums = targetPlot:FindFirstChild("AnimalPodiums")
	if not podiums then
		warn("[Spawn] AnimalPodiums não encontrado no plot")
		return false, "no podiums"
	end

	-- Sort podiums
	local podiumList = {}
	for _, child in ipairs(podiums:GetChildren()) do
		if child:IsA("Model") and tonumber(child.Name) then
			table.insert(podiumList, child)
		end
	end
	table.sort(podiumList, function(a,b) return tonumber(a.Name) < tonumber(b.Name) end)

	local spawnCFrame, spawnAttachment, claimObj, hitbox

	for _, podium in ipairs(podiumList) do
		local base = podium:FindFirstChild("Base")
		if base then
			local spawnPart = base:FindFirstChild("Spawn")
			if spawnPart then
				local occupied = false
				for _, obj in ipairs(spawnPart:GetChildren()) do
					if obj:IsA("Model") and obj.Name ~= "Attachment" then
						occupied = true; break
					end
				end
				if not occupied then
					spawnCFrame = spawnPart.CFrame * CFrame.new(0, -1.5, 0)
					spawnAttachment = spawnPart
					local claim = podium:FindFirstChild("Claim")
					if claim then
						local main = claim:FindFirstChild("Main")
						if main then claimObj = main end
						hitbox = claim:FindFirstChild("Hitbox")
					end
					break
				end
			end
		end
	end

	if not spawnCFrame then
		warn("[Spawn] Nenhum podium livre encontrado")
		return false, "no empty podium"
	end

	-- Clone and place
	local cloned = animalModel:Clone()
	cloned:PivotTo(spawnCFrame)
	cloned.Parent = workspace
	for _, desc in ipairs(cloned:GetDescendants()) do
		if desc:IsA("BasePart") then desc.Anchored = true end
	end

	-- Animation
	pcall(function()
		local animController = cloned:FindFirstChildOfClass("AnimationController")
		local animFolder = ReplicatedStorage:FindFirstChild("Animations")
		local animAnimals = animFolder and animFolder:FindFirstChild("Animals")
		local animalAnimFolder = animAnimals and animAnimals:FindFirstChild(animalName)
		local idleAnim = animalAnimFolder and animalAnimFolder:FindFirstChild("Idle")
		if animController and idleAnim then
			local animator = animController:FindFirstChildOfClass("Animator")
			if not animator then
				animator = Instance.new("Animator")
				animator.Parent = animController
			end
			local track = animator:LoadAnimation(idleAnim)
			track:Play()
		end
	end)

	-- Sound
	pcall(function()
		local animalSounds = ReplicatedStorage:FindFirstChild("Sounds")
		if animalSounds then
			local snd = animalSounds:FindFirstChild(animalName)
			if snd and snd:IsA("Sound") then
				local sndClone = snd:Clone()
				sndClone.Parent = cloned.PrimaryPart or cloned:FindFirstChildWhichIsA("BasePart")
				sndClone:Play()
				sndClone.Ended:Connect(function() sndClone:Destroy() end)
			end
		end
	end)

	-- Mutation
	applyMutationVisuals(cloned, mutation)

	-- Traits
	pcall(function()
		if traits and #traits > 0 and cloned.PrimaryPart then
			local modFolder = ReplicatedStorage:FindFirstChild("Models")
			local traitsModels = modFolder and modFolder:FindFirstChild("Traits")
			local vfxFolder = ReplicatedStorage:FindFirstChild("Vfx")
			local traitsVfx = vfxFolder and vfxFolder:FindFirstChild("Traits")
			local traitsPerAnimal = modFolder and modFolder:FindFirstChild("TraitsPerAnimal")
			for _, traitName in ipairs(traits) do
				local customPlacement = nil
				if traitsPerAnimal then
					local at = traitsPerAnimal:FindFirstChild(animalName)
					if at then customPlacement = at:FindFirstChild(traitName) end
				end
				if traitsVfx then
					local vfx = traitsVfx:FindFirstChild(traitName)
					if vfx then
						for _, vfxItem in ipairs(vfx:GetChildren()) do
							local vfxClone = vfxItem:Clone()
							if customPlacement and customPlacement:IsA("Attachment") then
								vfxClone.Parent = customPlacement
							elseif vfxClone:IsA("ParticleEmitter") or vfxClone:IsA("Beam") or vfxClone:IsA("Trail") then
								local att = cloned.PrimaryPart:FindFirstChildOfClass("Attachment") or Instance.new("Attachment")
								att.Parent = cloned.PrimaryPart
								vfxClone.Parent = att
							else
								vfxClone.Parent = cloned.PrimaryPart
							end
							if vfxClone:IsA("ParticleEmitter") or vfxClone:IsA("Trail") or vfxClone:IsA("Beam") then
								vfxClone.Enabled = true
							end
						end
					end
				end
				if traitsModels then
					local traitModel = traitsModels:FindFirstChild(traitName)
					if traitModel then
						local traitClone = traitModel:Clone()
						traitClone.Parent = cloned
						if traitClone:IsA("Model") then
							if customPlacement then
								if customPlacement:IsA("Attachment") then traitClone:PivotTo(customPlacement.WorldCFrame)
								elseif customPlacement:IsA("BasePart") then traitClone:PivotTo(customPlacement.CFrame) end
							elseif traitClone.PrimaryPart then
								traitClone:PivotTo(cloned.PrimaryPart.CFrame)
							end
							for _, part in ipairs(traitClone:GetDescendants()) do
								if part:IsA("BasePart") then
									part.Anchored = false; part.CanCollide = false; part.CanQuery = false
									local weld = Instance.new("WeldConstraint")
									weld.Part0 = cloned.PrimaryPart; weld.Part1 = part; weld.Parent = part
								end
							end
						elseif traitClone:IsA("BasePart") then
							traitClone.CFrame = cloned.PrimaryPart.CFrame
							traitClone.Anchored = false; traitClone.CanCollide = false; traitClone.CanQuery = false
							local weld = Instance.new("WeldConstraint")
							weld.Part0 = cloned.PrimaryPart; weld.Part1 = traitClone; weld.Parent = traitClone
						end
					end
				end
			end
		end
	end)

	-- Generation calc
	local totalGeneration = 0
	local generationRate = animalData.Generation or 0
	local traitBonus = 0
	if traits then
		for _, tname in ipairs(traits) do
			for _, data in pairs(traitMap) do
				if data.name == tname then traitBonus = traitBonus + generationRate * data.mult; break end
			end
		end
	end
	local mutationBonus = 0
	if mutation and mutation ~= "None" and mutationMap[mutation] then
		mutationBonus = generationRate * mutationMap[mutation].mult
	end
	local totalGenRate = generationRate + traitBonus + mutationBonus

	-- Overhead
	pcall(function()
		local overheadTemplate = ReplicatedStorage:FindFirstChild("Overheads") and ReplicatedStorage.Overheads:FindFirstChild("AnimalOverhead")
		local head = cloned:FindFirstChild("Head") or cloned.PrimaryPart
		if overheadTemplate and head then
			local overhead = overheadTemplate:Clone()
			overhead.StudsOffset = Vector3.new(0,7,0)
			overhead.Adornee = head
			overhead.Parent = head
			overhead.AlwaysOnTop = true
			overhead.MaxDistance = 200

			local mutationLabel = overhead:FindFirstChild("Mutation", true)
			if mutationLabel and mutationLabel:IsA("TextLabel") then
				if mutation and mutation ~= "None" then
					local mData = mutationMap[mutation]
					if mData then
						if mData.useRichText or mData.richText then
							mutationLabel.Text = mData.richText
							mutationLabel.RichText = true
						else
							mutationLabel.Text = mData.displayText or mutation
							mutationLabel.TextColor3 = mData.color or Color3.new(1,1,1)
						end
					end
					mutationLabel.Visible = true
				else
					mutationLabel.Visible = false
				end
			end

			local function updateOverhead(field, value)
				local f = overhead:FindFirstChild(field, true)
				if f then f.Text = value end
			end
			updateOverhead("DisplayName", animalData.DisplayName or animalName)
			updateOverhead("Rarity", animalData.Rarity or "Common")
			updateOverhead("Price", "$" .. formatNumber(animalData.Price or 0))
			updateOverhead("Generation", "$" .. formatNumber(totalGenRate) .. "/s")

			local rarityConfig = Rarities[animalData.Rarity]
			if rarityConfig then
				local rarityLabel = overhead:FindFirstChild("Rarity", true)
				if rarityLabel then
					if animalData.Rarity == "Brainrot God" then
						rarityLabel.TextColor3 = Color3.fromRGB(255,0,221)
					elseif animalData.Rarity == "Secret" then
						rarityLabel.TextColor3 = Color3.new(1,1,1)
					else
						rarityLabel.TextColor3 = rarityConfig.Color
					end
				end
			end

			local traitsContainer = overhead:FindFirstChild("Traits")
			if traitsContainer then
				traitsContainer.Visible = true
				local template = traitsContainer:FindFirstChild("Template")
				if traits and #traits > 0 and template then
					for _, tname in ipairs(traits) do
						local iconId = getTraitIconByName(tname)
						if iconId then
							local icon = template:Clone()
							icon.Image = iconId
							icon.Visible = true
							icon.Parent = traitsContainer
							icon.Name = tname .. "_Icon"
						end
					end
				end
			end
		end
	end)

	-- Attachment marker
	if spawnAttachment then
		local marker = Instance.new("Attachment")
		marker.Name = "Attachment"
		marker.Parent = spawnAttachment
	end

	-- Proximity prompt (sell)
	local primaryPart = cloned.PrimaryPart or cloned:FindFirstChildWhichIsA("BasePart")
	if primaryPart then
		local prompt = Instance.new("ProximityPrompt")
		prompt.ObjectText = animalData.DisplayName or animalName
		prompt.ActionText = "Sell: $" .. formatNumber((animalData.Price or 0) / 2)
		prompt.KeyboardKeyCode = Enum.KeyCode.E
		prompt.HoldDuration = 1.5
		prompt.MaxActivationDistance = 10
		prompt.Parent = primaryPart

		prompt.Triggered:Connect(function(player)
			if player ~= LocalPlayer then return end
			local sellPrice = (animalData.Price or 0) / 2
			pcall(function()
				local lb = PlayerGui:FindFirstChild("LeftBottom")
				local inner = lb and lb:FindFirstChild("LeftBottom")
				local currencyLabel = inner and inner:FindFirstChild("Currency")
				if currencyLabel and currencyLabel:IsA("TextLabel") then
					local cur = parseCurrency(currencyLabel.Text)
					currencyLabel.Text = "$" .. formatNumber(cur + sellPrice)
				end
			end)
			pcall(function()
				local ls = LocalPlayer:FindFirstChild("leaderstats")
				local cash = ls and ls:FindFirstChild("Cash")
				if cash then cash.Value = cash.Value + sellPrice end
			end)
			showCashout("+$" .. formatNumber(sellPrice), true, true)
			cloned:Destroy()
		end)
	end

	-- Generation loop
	if claimObj and hitbox then
		local collectLabel = claimObj:FindFirstChild("Collect")
		if collectLabel and collectLabel:IsA("TextLabel") then
			claimObj.Enabled = true
			collectLabel.RichText = true
			collectLabel.Text = 'Collect<br/><font color="#73ff00">$0</font>'

			if hitbox:IsA("BasePart") then
				hitbox.CanCollide = false
				hitbox.CanQuery = true
			end

			local lastCollectTime = 0
			hitbox.Touched:Connect(function(hit)
				local player = Players:GetPlayerFromCharacter(hit.Parent)
				if player == LocalPlayer and cloned and cloned.Parent then
					local now = os.time()
					if now - lastCollectTime >= 1 and totalGeneration > 0 then
						lastCollectTime = now
						pcall(function()
							local lb = PlayerGui:FindFirstChild("LeftBottom")
							local inner = lb and lb:FindFirstChild("LeftBottom")
							local currencyLabel = inner and inner:FindFirstChild("Currency")
							if currencyLabel and currencyLabel:IsA("TextLabel") then
								local cur = parseCurrency(currencyLabel.Text)
								currencyLabel.Text = "$" .. formatNumber(cur + totalGeneration)
							end
						end)
						pcall(function()
							local ls = LocalPlayer:FindFirstChild("leaderstats")
							local cash = ls and ls:FindFirstChild("Cash")
							if cash then cash.Value = cash.Value + totalGeneration end
						end)
						showCashout("+$" .. formatNumber(totalGeneration), true, true)
						totalGeneration = 0
						collectLabel.Text = 'Collect<br/><font color="#73ff00">$0</font>'
					end
				end
			end)

			claimObj:GetPropertyChangedSignal("Enabled"):Connect(function()
				if not claimObj.Enabled then claimObj.Enabled = true end
			end)

			task.spawn(function()
				while claimObj and claimObj.Parent and cloned and cloned.Parent do
					task.wait(1)
					totalGeneration = totalGeneration + totalGenRate
					collectLabel.Text = 'Collect<br/><font color="#73ff00">$' .. formatNumber(totalGeneration) .. "</font>"
				end
			end)
		end
	end

	table.insert(spawnedIllusions, cloned)
	print("[Spawn] " .. animalName .. " spawnado com sucesso!")
	return true, cloned
end

local function populateAnimalModal(filter)
	clearModalItems()
	local animalNames = {}
	for name in pairs(Animals) do table.insert(animalNames, name) end
	table.sort(animalNames)
	for _, name in ipairs(animalNames) do

			btn.MouseButton1Click:Connect(function()

	end
end

local function populateMutationModal(filter)
	clearModalItems()

		if not filter or filter == "" or mut:lower():find(filter:lower(), 1, true) then
			local iconId = (mut ~= "None" and mutationMap[mut]) and mutationMap[mut].icon or nil
			local btn = createModalItem(ModalScroll, mut, iconId)
			btn.MouseButton1Click:Connect(function()
				selectedMutation = mut

				ModalOverlay.Visible = false
			end)
		end
	end
end

local function populateTraitModal(filter)
	clearModalItems()
	for _, tname in ipairs(traitList) do
		if not filter or filter == "" or tname:lower():find(filter:lower(), 1, true) then
			local btn = createModalItem(ModalScroll, tname)
			-- restore selected state
			if table.find(selectedTraits, tname) then
				btn.BackgroundColor3 = Color3.new(0.078,0.392,0.078)
			end
			btn.MouseButton1Click:Connect(function()
				local idx = table.find(selectedTraits, tname)
				if idx then
					table.remove(selectedTraits, idx)
					btn.BackgroundColor3 = Color3.new(0.059,0.059,0.059)
				else
					table.insert(selectedTraits, tname)
					btn.BackgroundColor3 = Color3.new(0.078,0.392,0.078)
				end
				traitsBtn.Text = "Traits: " .. #selectedTraits .. " Selected"
			end)
		end
	end
end

local function showModal(modalType)
	currentModalType = modalType
	ModalOverlay.Visible = true
	ModalSearch.Text = ""

	if modalType == "animal" then
		ModalTitle.Text = "SELECIONAR ANIMAL"
		populateAnimalModal()
	elseif modalType == "mutation" then
		ModalTitle.Text = "SELECIONAR MUTATION"
		populateMutationModal()
	elseif modalType == "trait" then
		ModalTitle.Text = "SELECIONAR TRAITS"
		populateTraitModal()
	end

	if modalSearchConn then modalSearchConn:Disconnect() end
	modalSearchConn = ModalSearch:GetPropertyChangedSignal("Text"):Connect(function()
		local f = ModalSearch.Text
		if currentModalType == "animal" then populateAnimalModal(f)
		elseif currentModalType == "mutation" then populateMutationModal(f)
		elseif currentModalType == "trait" then populateTraitModal(f) end
	end)
end

local function findTradeGui()

local function tradeLastPlayer()
	local targetName = otherNameBox.Text
	if not targetName or targetName == "" then
		warn("[Trade] Nome do player vazio")
		return
	end
	pcall(function()
		local t = findTradeGui()
		if not t then
			warn("[Trade] Trade GUI não encontrada")
			return
		end

		-- Update signs
		local yourSign = t:FindFirstChild("YourSign")
		local otherSign = t:FindFirstChild("OtherSign")
		if yourSign then
			local uLabel = yourSign:FindFirstChild("Username") or yourSign:FindFirstChildOfClass("TextLabel")
			if uLabel then uLabel.Text = "Sua oferta" end
			local img = yourSign:FindFirstChild("PlayerImage")
			if img then
				local hs = img:FindFirstChild("Headshot")
				if hs then hs.Image = "rbxthumb://type=AvatarHeadShot&id=" .. LocalPlayer.UserId .. "&w=100&h=100" end
			end
		end
		if otherSign then
			local uLabel = otherSign:FindFirstChild("Username") or otherSign:FindFirstChildOfClass("TextLabel")
			if uLabel then uLabel.Text = targetName .. "'s offer" end
		end

		-- Update slot counts
		local yourPanel  = t:FindFirstChild("Your")
		local otherPanel = t:FindFirstChild("Other")
		if yourPanel then
			local slots = yourPanel:FindFirstChild("BaseSlots") or yourPanel:FindFirstChildOfClass("TextLabel")
			if slots then slots.Text = "0/" .. yourSlotsBox.Text end
			local ready = yourPanel:FindFirstChild("Ready")
			if ready then ready.Visible = false end
		end
		if otherPanel then
			local slots = otherPanel:FindFirstChild("BaseSlots") or otherPanel:FindFirstChildOfClass("TextLabel")
			if slots then slots.Text = "0/" .. otherSlotsBox.Text end
			local ready = otherPanel:FindFirstChild("Ready")
			if ready then ready.Visible = false end
			local readyBtn = otherPanel:FindFirstChild("ReadyButton")
			if readyBtn then
				readyBtn.BackgroundColor3 = Color3.new(0.439,0.439,0.439)
				local txt = readyBtn:FindFirstChild("Txt") or readyBtn:FindFirstChild("Text")
				if txt then txt.Text = "READY" end
			end
		end

		local timer = t:FindFirstChild("Timer")
		if timer then timer.Text = "⏰5.5s Left" end

		-- Countdown animation
		task.spawn(function()
			local countdown = 5.5
			while countdown > 0 do
				if timer then timer.Text = "⏰" .. string.format("%.1f", countdown) .. "s Left" end
				task.wait(0.1)
				countdown = countdown - 0.1
			end
			if timer then timer.Text = "Processando..." end
			task.wait(0.5)

			if otherPanel then
				local readyBtn = otherPanel:FindFirstChild("ReadyButton")
				if readyBtn then
					readyBtn.BackgroundColor3 = Color3.new(0.318,0.620,0.337)
					local txt = readyBtn:FindFirstChild("Txt") or readyBtn:FindFirstChild("Text")
					if txt then txt.Text = "CONFIRM" end
				end
				local readyFrame = otherPanel:FindFirstChild("Ready")
				if readyFrame then
					local native = readyFrame:FindFirstChild("NativeReady")
					if native then
						local clone = native:Clone()
						clone.Name = "CustomReady"
						clone.Visible = true
						clone.Parent = otherPanel
						local lbl = clone:FindFirstChild("Label") or clone:FindFirstChildOfClass("TextLabel")
						if lbl then lbl.Text = "Confirmado!" end
					end
				end
			end

			-- Success notification
			task.wait(1)
			local notifGui = PlayerGui:FindFirstChild("Notification")
			local notifInner = notifGui and notifGui:FindFirstChild("Notification")
			local notifTemplate = notifInner and notifInner:FindFirstChild("Template")
			if notifTemplate then
				local notif = notifTemplate:Clone()
				notif.RichText = true
				notif.Text = '<font color="#92FF67">Trade com @' .. targetName .. ' completado! 🎉</font>'
				notif.Visible = true
				notif.Parent = notifInner
				task.delay(5, function() if notif and notif.Parent then notif:Destroy() end end)
			else
				-- fallback simple notif
				showCashout("Trade com " .. targetName .. " OK!", true, false)
			end
			print("[Trade] Trade completado com " .. targetName)
		end)
	end)
end

-- // KEYBIND REBINDING // --
local currentKeybinds = {
	toggle = Enum.KeyCode.RightShift,
	spawn  = Enum.KeyCode.T,
	trade  = Enum.KeyCode.Y,
}
local rebindTarget = nil

local function setRebindTarget(btn, keyName)
	if rebindTarget then rebindTarget.BackgroundColor3 = Color3.new(0.137,0.137,0.137) end
	rebindTarget = btn
	rebindTarget.BackgroundColor3 = Color3.fromRGB(180,120,0)
	rebindTarget.Text = "..."
	-- listen once
	local conn; conn = UserInputService.InputBegan:Connect(function(input, gp)
		if gp then return end
		if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
		conn:Disconnect()
		currentKeybinds[keyName] = input.KeyCode
		rebindTarget.Text = input.KeyCode.Name
		rebindTarget.BackgroundColor3 = Color3.new(0.137,0.137,0.137)
		rebindTarget = nil
	end)
end

toggleKeyBtn.MouseButton1Click:Connect(function() setRebindTarget(toggleKeyBtn, "toggle") end)
spawnKeyBtn.MouseButton1Click:Connect(function()  setRebindTarget(spawnKeyBtn, "spawn") end)
tradeKeyBtn.MouseButton1Click:Connect(function()  setRebindTarget(tradeKeyBtn, "trade") end)

-- // KEYBINDS // --
UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if rebindTarget then return end -- skip while rebinding
	if input.KeyCode == currentKeybinds.toggle then
		Main.Visible = not Main.Visible
	elseif input.KeyCode == currentKeybinds.spawn then
		spawnBtn:Invoke() -- triggers same as button click
		local targetName = nil
		if targetMode == "HIM" then targetName = otherNameBox.Text end
		spawnAnimal(selectedAnimal, selectedTraits, selectedMutation ~= "None" and selectedMutation or nil, targetName)
	elseif input.KeyCode == currentKeybinds.trade then
		tradeLastPlayer()
	end
end)

-- // DRAGGING // --
local dragging = false
local dragStartPos = nil
local dragStartMain = nil
TitleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStartPos = input.Position
		dragStartMain = Main.Position
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStartPos
		Main.Position = UDim2.new(
			dragStartMain.X.Scale, dragStartMain.X.Offset + delta.X,
			dragStartMain.Y.Scale, dragStartMain.Y.Offset + delta.Y
		)
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)
