-- Format number helper
local function formatNumber(num)
	local suffixes = {"", "K", "M", "B", "T", "Q"}
	local tier = 1
	while num >= 1000 and tier < #suffixes do
		num = num / 1000
		tier = tier + 1
	end
	if num % 1 == 0 then
		return string.format("%d%s", num, suffixes[tier])
	else
		return string.format("%.1f%s", num, suffixes[tier])
	end
end

local function parseCurrency(str)
	str = str:gsub("<[^>]+>", ""):gsub("%$", ""):gsub(",", ""):gsub("[^%d%.%a]", "")
	local numStr, suffix = str:match("^(%d*%.?%d+)(%a*)")
	local num = tonumber(numStr) or 0
	suffix = suffix:upper()
	local multipliers = {K = 1000, M = 1000000, B = 1000000000, T = 1000000000000, Q = 1000000000000000}
	return num * (multipliers[suffix] or 1)
end

		local ls = LocalPlayer:FindFirstChild("leaderstats")
		local cash = ls and ls:FindFirstChild("Cash")
		if cash then cash.Value = cash.Value + sellPrice end
		showCashout("+$" .. formatNumber(sellPrice), true, true)
		cloned:Destroy()
		marker:Destroy()
	end)

	table.insert(spawnedIllusions, cloned)
	return true, cloned
end

local function populateAnimalModal(filter)
	for _, child in ipairs(ModalScroll:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end
	local animalNames = {}
	for name in pairs(Animals) do table.insert(animalNames, name) end
	table.sort(animalNames)
	for _, name in ipairs(animalNames) do
		if not filter or string.find(name:lower(), filter:lower(), 1, true) then
			local btn = createModalItem(ModalScroll, name)
			btn.MouseButton1Click:Connect(function()
				selectedAnimal = name
				animalBtn.Text = "Animal: " .. name
				ModalOverlay.Visible = false
			end)
		end
	end
end

local function populateMutationModal(filter)
	for _, child in ipairs(ModalScroll:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end
	for _, mut in ipairs(mutationList) do
		if not filter or string.find(mut:lower(), filter:lower(), 1, true) then
			local iconId = nil
			local name = mut
			if mut ~= "None" and mutationMap[mut] then iconId = mutationMap[mut].icon end
			local btn = createModalItem(ModalScroll, name, iconId)
			btn.MouseButton1Click:Connect(function()
				selectedMutation = mut
				mutationBtn.Text = "Mutation: " .. name
				ModalOverlay.Visible = false
			end)
		end
	end
end

local function populateTraitModal(filter)
	for _, child in ipairs(ModalScroll:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end
	for _, tname in ipairs(traitList) do
		if not filter or string.find(tname:lower(), filter:lower(), 1, true) then
			local btn = createModalItem(ModalScroll, tname)
			btn.MouseButton1Click:Connect(function()
				local idx = table.find(selectedTraits, tname)
				if idx then
					table.remove(selectedTraits, idx)
					btn.BackgroundColor3 = Color3.new(0.0588235, 0.0588235, 0.0588235)
				else
					table.insert(selectedTraits, tname)
					btn.BackgroundColor3 = Color3.new(0.0784314, 0.392157, 0.0784314)
				end
				traitsBtn.Text = "Traits: " .. #selectedTraits .. " Selected"
			end)
		end
	end
end

local function showModal(type, callback)
	currentModalType = type
	ModalOverlay.Visible = true
	ModalSearch.Text = ""
	ModalConfirmBtn.Visible = false
	if type == "animal" then
		ModalTitle.Text = "SELECT ANIMAL"
		ModalGrid.CellSize = UDim2.new(0, 215, 0, 30)
		populateAnimalModal()
	elseif type == "mutation" then
		ModalTitle.Text = "SELECT MUTATION"
		ModalGrid.CellSize = UDim2.new(0, 215, 0, 30)
		populateMutationModal()
	elseif type == "trait" then
		ModalTitle.Text = "SELECT TRAITS"
		ModalGrid.CellSize = UDim2.new(0, 215, 0, 30)
		populateTraitModal()
	end
	if modalSearchConn then
	modalSearchConn:Disconnect()
end

	local filter = ModalSearch.Text

	if currentModalType == "animal" then
		populateAnimalModal(filter)

	elseif currentModalType == "mutation" then
		populateMutationModal(filter)

	elseif currentModalType == "trait" then
		populateTraitModal(filter)
	end
end)

local function tradeLastPlayer()
	local targetName = otherNameBox.Text
	if targetName == "" then return end
	local tradeGui = PlayerGui:WaitForChild("TradeLiveTrade", 1)
	if not tradeGui then return end
	local t = tradeGui:WaitForChild("TradeLiveTrade", 1)
	if not t then return end

	-- Set up fake UI
	t.YourSign.Username.Text = "Your offer"
	t.OtherSign.Username.Text = targetName .. "'s offer"
	local defaultHead = "rbxthumb://type=AvatarHeadShot&id=47528603&w=100&h=100"
	t.YourSign.PlayerImage.Headshot.Image = "rbxthumb://type=AvatarHeadShot&id=" .. LocalPlayer.UserId .. "&w=100&h=100"
	t.OtherSign.PlayerImage.Headshot.Image = defaultHead
	t.Your.BaseSlots.Text = "0/" .. yourSlotsBox.Text
	t.Other.BaseSlots.Text = "0/" .. otherSlotsBox.Text
	t.Your.Ready.Visible = false
	t.Other.Ready.Visible = false
	t.Other.ReadyButton.BackgroundColor3 = Color3.new(0.439216, 0.439216, 0.439216)
	t.Other.ReadyButton.Txt.Text = "READY"
	t.YourSign.Buttons.Frame.TextButton.TextLabel.Text = "🤝 Deal?"
	t.Timer.Text = "⏰5.5s Left"

	-- Animate countdown and completion
	task.spawn(function()
		local timer = 5.5
		while timer > 0 do
			t.Timer.Text = "⏰" .. string.format("%.1f", timer) .. "s Left"
			task.wait(0.1)
			timer = timer - 0.1
		end
		t.Timer.Text = "Processing..."
		task.wait(0.5)
		t.Other.ReadyButton.BackgroundColor3 = Color3.new(0.317647, 0.619608, 0.337255)
		t.Other.ReadyButton.Txt.Text = "CONFIRM"
		local customReady = t.Other.Ready:FindFirstChild("CustomReady") or t.Other.Ready.NativeReady:Clone()
		customReady.Name = "CustomReady"
		customReady.Visible = true
		customReady.Parent = t.Other
		local label = customReady:FindFirstChild("Label")
		if label then label.Text = "Confirmed!" end
		task.wait(1)
		-- Success sound
		local sfx = ReplicatedStorage:FindFirstChild("Sounds")
		if sfx then
			local successSound = sfx.Sfx:FindFirstChild("Success")
			if successSound then
				local s = successSound:Clone()
				s.Parent = workspace
				s:Play()
				s.Ended:Connect(function() s:Destroy() end)
			end
		end
		local notifTemplate = PlayerGui.Notification.Notification:FindFirstChild("Template")
		if notifTemplate then
			local notif = notifTemplate:Clone()
			notif.RichText = true
			notif.Text = "<font color=\"#92FF67\">Trade with @" .. targetName .. " completed! 🎉</font>"
			notif.Visible = true
			notif.Parent = PlayerGui.Notification.Notification
			task.delay(5, function() notif:Destroy() end)
		end
	end)
end

-- Dragging
local dragging = false
local dragStartInput = nil
local startPos = nil
TitleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStartInput = input.Position
		startPos = Main.Position
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		local delta = input.Position - dragStartInput
		Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)
